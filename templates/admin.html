<script>

const KEY = {{ admin_key|tojson }};


async function cargar() {

  const [pf, pm] = await Promise.all([

    fetch('/api/photos', {cache:'no-store'}).then(r => r.json()),

    fetch('/api/mensajes', {cache:'no-store'}).then(r => r.json())

  ]);


  const cf = document.getElementById('fotos');

  cf.innerHTML = '';

  if (!pf.photos.length) cf.innerHTML = '<p class="vacio">Sin fotos.</p>';

  pf.photos.forEach(src => {

    const nombre = src.split('/').pop();

    const d = document.createElement('div');

    d.className = 'item';

    const img = document.createElement('img');

    img.src = src;

    const b = document.createElement('button');

    b.className = 'x';

    b.textContent = 'X';

    b.onclick = () => borrar('/api/admin/borrar-foto', {nombre});

    d.append(img, b);

    cf.appendChild(d);

  });


  const cm = document.getElementById('mensajes');

  cm.innerHTML = '';

  if (!pm.mensajes.length) cm.innerHTML = '<p class="vacio">Sin saludos.</p>';

  pm.mensajes.forEach(m => {

    const d = document.createElement('div');

    d.className = 'msg';

    const izq = document.createElement('div');

    const t = document.createElement('div');

    t.textContent = m.texto;

    izq.appendChild(t);

    if (m.nombre) {

      const a = document.createElement('div');

      a.className = 'autor';

      a.textContent = '- ' + m.nombre;

      izq.appendChild(a);

    }

    const b = document.createElement('button');

    b.textContent = 'Borrar';

    b.onclick = () => borrar('/api/admin/borrar-mensaje', {ts: m.ts});

    d.append(izq, b);

    cm.appendChild(d);

  });

}


async function borrar(url, datos) {

  if (!confirm('Borrar definitivamente?')) return;

  const fd = new FormData();

  fd.append('key', KEY);

  for (const k in datos) fd.append(k, datos[k]);

  await fetch(url, {method:'POST', body: fd});

  cargar();

}


cargar();

setInterval(cargar, 10000);
